# Tests that fail because of a known bug, left unfixed on purpose. They still
# run and are reported, but bin/test (and so the PR gate) doesn't fail on them.
# Any other failure fails it. When a listed test passes, bin/test warns: the
# bug is fixed (or doesn't show here), and the line can go.
#
# One test per line: <suite> <test>, as the TAP report names it, then the bug.

# buffer-tools: :BufferClose shows the previous buffer by number, not the most
# recently used one (autoload/buffer_tools.vim, buffer_tools#close)
buffer-tools close_shows_the_last_used
# buffer-tools: :BufferClose with only terminal/netrw/nofile buffers left
# raises E684 instead of leaving an empty buffer
buffer-tools close_with_only_a_terminal_left
# buffer-tools: :BufferClose uses bdelete!, so unsaved changes are lost
buffer-tools close_keeps_unsaved_changes

# glow-preview: live mode run again opens another tmux pane instead of
# replacing the previous one
glow-preview live_preview_again_replaces_the_pane

# dispatch-extras: the repeated :Start timed out once on CI (fzf-utils#3's run,
# 2026-11-08) and passed on the other runs; cause not known
dispatch-extras repeat_start_reruns_the_last_start

# gh-extras: 'aboveleft 10new' above the thread list is evened out by
# 'equalalways' with the window above it, so the reply draft isn't 10 lines
# (13 in a 35-line Vim; autoload/gh_extras/threads.vim, reply_draft)
gh-extras reply_draft_is_10_lines
# gh-extras: <CR> on a whole-file thread opens the commit it was made on, even
# on the PR's own branch, where a thread opens the work tree
# (autoload/gh_extras/threads.vim, s:outdated counts a null line as outdated)
gh-extras whole_file_thread_on_its_branch_opens_the_work_tree

# pytest.vim: --trace sessions run with :Start! -strategy=terminal, but
# vim-dispatch has no -strategy option: they're in a Vim terminal only while
# g:dispatch_no_tmux_start is set (the dotfiles' default). Toggled off (_cs),
# they go to tmux, or outside it nowhere visible (autoload/pytest/dispatch.vim)
pytest.vim trace_in_a_terminal_whatever_the_start_strategy
