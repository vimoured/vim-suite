# Tests that fail because of a known bug, left unfixed on purpose. They still
# run and are reported, but bin/test (and so the PR gate) doesn't fail on them.
# Any other failure fails it. When a listed test passes, bin/test warns: the
# bug is fixed (or doesn't show here), and the line can go.
#
# One test per line: <suite> <test>, as the TAP report names it, then the bug.

# pytest.vim: :YankTest* sets @+, which errors (E354) on a Vim without
# +clipboard, as on CI. autoload/pytest.vim:43; fix: pytest-clipboard.patch
pytest.vim file_keys
pytest.vim list_keys
pytest.vim list_entry_moved_or_gone
pytest.vim outline

# buffer-tools: :BufferClose shows the previous buffer by number, not the most
# recently used one (autoload/buffer_tools.vim, buffer_tools#close)
buffer-tools close_shows_the_last_used
# buffer-tools: :BufferClose with only terminal/netrw/nofile buffers left
# raises E684 instead of leaving an empty buffer
buffer-tools close_with_only_a_terminal_left
# buffer-tools: :BufferClose uses bdelete!, so unsaved changes are lost
buffer-tools close_keeps_unsaved_changes
